""" Voxify - Text to Audio API Converts text into MP3 audio using Microsoft Edge TTS. """ import os import tempfile from pathlib import Path import edge_tts from fastapi import APIRouter, HTTPException from fastapi.responses import FileResponse from pydantic import BaseModel, Field # ========================================== # Router # ========================================== router = APIRouter( prefix="/text-to-audio", tags=["Text to Audio"], ) # ========================================== # Supported Voices # ========================================== VOICE_MAP = { "en-IN": "en-IN-NeerjaNeural", "hi-IN": "hi-IN-SwaraNeural", "gu-IN": "gu-IN-DhwaniNeural", "mr-IN": "mr-IN-AarohiNeural", "ta-IN": "ta-IN-PallaviNeural", } # ========================================== # Request Model # ========================================== class TextToAudioRequest(BaseModel): text: str = Field( ..., min_length=1, max_length=5000, ) language: str = Field( ..., description="Language code such as en-IN, hi-IN, gu-IN, mr-IN, ta-IN", ) # ========================================== # Text → Audio # ========================================== @router.post("/generate") async def generate_audio( request: TextToAudioRequest, ): text = request.text.strip() language = request.language.strip() # -------------------------------------- # Validate text # -------------------------------------- if not text: raise HTTPException( status_code=400, detail="Text cannot be empty.", ) # -------------------------------------- # Validate language # -------------------------------------- if language not in VOICE_MAP: raise HTTPException( status_code=400, detail=( f"Unsupported language: {language}. " f"Supported languages: {', '.join(VOICE_MAP.keys())}" ), ) voice = VOICE_MAP[language] # -------------------------------------- # Temporary MP3 file # -------------------------------------- temp_file = tempfile.NamedTemporaryFile( suffix=".mp3", delete=False, ) output_path = Path(temp_file.name) temp_file.close() try: # ---------------------------------- # Generate speech # ---------------------------------- communicate = edge_tts.Communicate( text, voice, ) await communicate.save(str(output_path)) # ---------------------------------- # Verify generated file # ---------------------------------- if not output_path.exists(): raise RuntimeError( "Audio file was not generated." ) if output_path.stat().st_size == 0: raise RuntimeError( "Generated audio file is empty." ) # ---------------------------------- # Return MP3 # ---------------------------------- return FileResponse( path=str(output_path), media_type="audio/mpeg", filename="voxify-text-to-audio.mp3", background=None, ) except Exception as error: # Clean up temporary file if output_path.exists(): try: os.remove(output_path) except OSError: pass raise HTTPException( status_code=500, detail=f"Text to Audio generation failed: {str(error)}", )
